## 获取系统配置

<GET>/config/system</GET>

### 响应

<TypeTable
  noRequiredMark={true}
  type={{
    timezone: {
      description: '系统时区，IANA 时区标识符（如 Asia/Shanghai）。重启服务后生效',
      type: 'string',
      required: true,
    },
    npmRegistry: {
      description: 'npm 镜像源地址（完整 URL），留空使用 npm 官方源',
      type: 'string',
      required: true,
    },
    pipIndexUrl: {
      description: 'pip index-url（完整 URL），留空使用 PyPI 官方源',
      type: 'string',
      required: true,
    },
    aptMirrorUrl: {
      description: 'APT 镜像源地址，填写 Debian 主源根路径（完整 URL，如 https://mirrors.aliyun.com/debian）',
      type: 'string',
      required: true,
    },
    logRetentionDays: {
      description: '系统日志保留天数（含操作日志和登录日志）',
      type: 'string<number>',
      required: true,
    },
    messageRetentionDays: {
      description: '消息中心已读消息保留天数',
      type: 'string<number>',
      required: true,
    },
    taskHistoryRetentionDays: {
      description: '定时任务执行统计数据保留天数',
      type: 'string<number>',
      required: true,
    },
    cleanupCronExpression: {
      description: '定时自动清理的 cron 表达式，控制清理任务的执行频率。为空时系统自动生成每天执行一次的默认表达式',
      type: 'string',
      required: true,
    },
    cleanupCronEnabled: {
      description: '是否启用定时自动清理',
      typeDescription: '"true" 启用，"false" 禁用',
      type: 'string',
      required: true,
    },
  }}
/>

```json title="示例"
{
  "timezone": "Asia/Shanghai",
  "npmRegistry": "https://registry.npmmirror.com",
  "pipIndexUrl": "https://mirrors.aliyun.com/pypi/simple/",
  "aptMirrorUrl": "http://mirrors.aliyun.com/debian",
  "logRetentionDays": "7",
  "messageRetentionDays": "7",
  "taskHistoryRetentionDays": "7",
  "cleanupCronExpression": "23 4 * * *",
  "cleanupCronEnabled": "true"
}
```

## 更新系统配置

<POST>/config/system</POST>

支持部分字段更新，仅传入需要修改的字段即可。

### 请求

<TypeTable
  type={{
    timezone: {
      description: '系统时区，IANA 时区标识符（注：重启服务后生效）',
      type: 'string',
    },
    npmRegistry: {
      description: 'npm 镜像源地址（完整 URL）',
      type: 'string',
    },
    pipIndexUrl: {
      description: 'pip index-url（完整 URL）',
      type: 'string',
    },
    aptMirrorUrl: {
      description: 'APT 镜像源地址，Debian 主源根路径',
      type: 'string',
    },
    logRetentionDays: {
      description: '系统日志保留天数（正整数字符串）',
      type: 'string<number>',
    },
    messageRetentionDays: {
      description: '消息中心已读消息保留天数（正整数字符串）',
      type: 'string<number>',
    },
    taskHistoryRetentionDays: {
      description: '定时任务执行统计数据保留天数（正整数字符串）',
      type: 'string<number>',
    },
    cleanupCronExpression: {
      description: '定时自动清理的 cron 表达式',
      type: 'string',
    },
    cleanupCronEnabled: {
      description: '是否启用定时自动清理',
      typeDescription: '"true" 启用，"false" 禁用',
      type: 'string',
    },
  }}
/>
