title: 更新已读状态
---
import { PUT } from '@/components/RequestTag';

支持批量操作

<PUT>/message/status</PUT>

## 请求

<TypeTable
  type={{
    id: {
      description: '唯一标识',
      type: 'number | number[]',
      required: true,
    },
    status: {
      description: '新的状态',
      type: 'number<1 | 0>',
      typeDescription: '1 已读，0 未读',
      required: true,
    },
  }}
/>

## 更新全部消息

<PUT>/message/status/all</PUT>

### 请求

<TypeTable
  type={{
    status: {
      description: '目标已读状态值',
      type: 'number<1 | 0>',
      default: 1,
    },
  }}
/>
