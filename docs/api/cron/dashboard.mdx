## 获取指标数据

<GET>/cron/dashboard/stats</GET>

### 请求

<TypeTable
  type={{
    taskType: {
      description: '任务类型过滤',
      type: "string<'all' | 'user' | 'system'>",
      default: 'all',
    },
  }}
/>

### 响应

<TypeTable
  noRequiredMark={true}
  type={{
    todaySuccessCount: {
      description: '今日成功执行的任务数量',
      type: 'number',
      required: true,
    },
    todayFailureCount: {
      description: '今日执行失败的任务数量',
      type: 'number',
      required: true,
    },
    yesterdaySuccessCount: {
      description: '昨日成功执行的任务数量',
      type: 'number',
      required: true,
    },
    yesterdayFailureCount: {
      description: '昨日执行失败的任务数量',
      type: 'number',
      required: true,
    },
    enabledCount: {
      description: '当前启用的任务数量',
      type: 'number',
      required: true,
    },
    disabledCount: {
      description: '当前禁用的任务数量',
      type: 'number',
      required: true,
    },
    runningCount: {
      description: '当前正在运行的任务数量',
      type: 'number',
      required: true,
    },
  }}
/>

```json title="示例"
{
    "todaySuccessCount": 8,
    "todayFailureCount": 0,
    "yesterdaySuccessCount": 13,
    "yesterdayFailureCount": 1,
    "enabledCount": 18,
    "disabledCount": 23,
    "runningCount": 0
}
```

## 获取任务运行趋势

<GET>/cron/dashboard/trend</GET>

### 请求

<TypeTable
  type={{
    taskType: {
      description: '任务类型过滤',
      type: "string<'all' | 'user' | 'system'>",
      default: 'all',
    },
    date: {
      description: '日期过滤，默认不填为今天',
      type: "string<'yyyy-MM-dd'>",
      default: '今天',
    },
  }}
/>

### 响应

返回对象，包含趋势数据数组和耗时排行数据

<TypeTable
  noRequiredMark={true}
  type={{
    trend: {
      description: '任务执行趋势数据数组',
      type: 'array',
      required: true,
    },
    'trend[].timestamp': {
      description: '执行时间戳',
      type: 'number',
      required: true,
    },
    'trend[].taskId': {
      description: '任务ID',
      type: 'number',
      required: true,
    },
    'trend[].taskName': {
      description: '任务名称',
      type: 'string',
      required: true,
    },
    'trend[].taskType': {
      description: '任务类型',
      type: "string<'system' | 'user'>",
      required: true,
    },
    'trend[].duration': {
      description: '运行时长，单位毫秒',
      type: 'number',
      required: true,
    },
    'trend[].success': {
      description: '运行结果（是否成功）',
      type: 'boolean',
      required: true,
    },
    durationRanking: {
      description: '耗时排行数据',
      type: 'object',
      required: true,
    },
    'durationRanking.longest': {
      description: '最长耗时 Top 5（按 duration 降序）',
      type: 'array',
      required: true,
    },
    'durationRanking.shortest': {
      description: '最短耗时 Top 5（按 duration 升序）',
      type: 'array',
      required: true,
    },
    'durationRanking.longest[].taskId': {
      description: '任务ID',
      type: 'number',
      required: true,
    },
    'durationRanking.longest[].taskName': {
      description: '任务名称',
      type: 'string',
      required: true,
    },
    'durationRanking.longest[].taskType': {
      description: '任务类型',
      type: "string<'system' | 'user'>",
      required: true,
    },
    'durationRanking.longest[].execTimestamp': {
      description: '执行时间戳（13位毫秒）',
      type: 'number',
      required: true,
    },
    'durationRanking.longest[].duration': {
      description: '运行时长，单位毫秒',
      type: 'number',
      required: true,
    },
    'durationRanking.longest[].success': {
      description: '运行结果（是否成功）',
      type: 'boolean',
      required: true,
    },
  }}
/>

> `durationRanking.shortest[]` 的字段结构与 `durationRanking.longest[]` 完全一致。

```json title="示例"
{
    "trend": [
        {
            "timestamp": 1773161661000,
            "taskId": 1,
            "taskName": "更新仓库",
            "taskType": "user",
            "duration": 21749,
            "success": true
        },
        {
            "timestamp": 1773233647000,
            "taskId": 1,
            "taskName": "更新仓库",
            "taskType": "user",
            "duration": 7042,
            "success": true
        }
    ],
    "durationRanking": {
        "longest": [
            {
                "taskId": 1,
                "taskName": "更新仓库",
                "taskType": "user",
                "execTimestamp": 1773161661000,
                "duration": 21749,
                "success": true
            }
        ],
        "shortest": [
            {
                "taskId": 1,
                "taskName": "更新仓库",
                "taskType": "user",
                "execTimestamp": 1773233647000,
                "duration": 7042,
                "success": true
            }
        ]
    }
}
```

## 获取正在运行中的任务

<GET>/cron/dashboard/running</GET>

### 请求

<TypeTable
  type={{
    taskType: {
      description: '任务类型过滤',
      type: "string<'all' | 'user' | 'system'>",
      default: 'all',
    },
  }}
/>

### 响应

返回数组对象

<TypeTable
  noRequiredMark={true}
  type={{
    id: {
      description: '任务ID',
      type: 'number',
      required: true,
    },
    name: {
      description: '任务名称',
      type: 'string',
      required: true,
    },
    type: {
      description: '任务类型',
      type: "string<'system' | 'user'>",
      required: true,
    },
    cron: {
      description: '任务的cron表达式',
      type: 'string',
      required: true,
    },
    shell: {
      description: '任务的执行命令',
      type: 'string',
      required: true,
    },
  }}
/>

```json title="示例"
[
    {
        "id": 1,
        "name": "更新仓库",
        "type": "user",
        "cron": "54 */2 * * *",
        "shell": "arcadia update repo"
    },
    {
        "id": 2,
        "name": "清理日志",
        "type": "system",
        "cron": "48 5 * * *",
        "shell": "arcadia rmlog 7"
    }
    ...
]
